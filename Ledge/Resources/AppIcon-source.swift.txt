import AppKit

// Ledge 앱 아이콘 원본 (2026-10-31 확정: "ㄴ자 벽 선반에 떨어뜨리기", 메뉴바 기호와 같은 모티브).
// 실행: swiftc -O -o gen AppIcon-source.swift && ./gen <AppIcon.appiconset 폴더>
// 128px 이상: drawV1(그림자 있음). 64px 이하: drawVSmall(그림자 없음, 더 굵게).

func rgb(_ r: CGFloat, _ g: CGFloat, _ b: CGFloat) -> NSColor { NSColor(red: r, green: g, blue: b, alpha: 1) }

func squircle(_ size: CGFloat) -> (NSRect, NSBezierPath, CGFloat) {
    let s = size / 1024
    let inset = 100 * s
    let rect = NSRect(x: inset, y: inset, width: size - inset * 2, height: size - inset * 2)
    return (rect, NSBezierPath(roundedRect: rect, xRadius: 185 * s, yRadius: 185 * s), s)
}

func withShadow(_ alpha: CGFloat, blur: CGFloat, dy: CGFloat, _ body: () -> Void) {
    let sh = NSShadow()
    sh.shadowColor = NSColor.black.withAlphaComponent(alpha)
    sh.shadowBlurRadius = blur
    sh.shadowOffset = NSSize(width: 0, height: dy)
    NSGraphicsContext.saveGraphicsState(); sh.set(); body(); NSGraphicsContext.restoreGraphicsState()
}

let orangeTop = rgb(1.00, 0.66, 0.28), orangeBot = rgb(0.95, 0.45, 0.16)
func bgOrange(_ size: CGFloat, shadow: Bool = true) -> (NSRect, NSBezierPath, CGFloat) {
    let (rect, bg, s) = squircle(size)
    if shadow { withShadow(0.25, blur: 24 * s, dy: -12 * s) { rgb(0.98, 0.55, 0.20).setFill(); bg.fill() } } else { rgb(0.98, 0.55, 0.20).setFill(); bg.fill() }
    NSGraphicsContext.saveGraphicsState(); bg.addClip()
    NSGradient(colors: [orangeTop, orangeBot])!.draw(in: rect, angle: -90)
    return (rect, bg, s)
}
/// ㄴ자 벽 선반 (흰색). 왼쪽 벽 + 아래 선반. 반환: 선반 윗면 y, 벽 오른쪽 x
@discardableResult
func wallShelf(_ rect: NSRect, _ s: CGFloat, thick: CGFloat = 80, shadow: Bool = true) -> (CGFloat, CGFloat) {
    let x0 = rect.minX + 130 * s, y0 = rect.minY + 150 * s
    let w = rect.width - 260 * s, h = rect.height - 300 * s
    let p = NSBezierPath()
    let t = thick * s, r = 28 * s
    // ㄴ 외곽을 하나의 패스로
    p.move(to: NSPoint(x: x0 + r, y: y0))
    p.line(to: NSPoint(x: x0 + w - r, y: y0)); p.appendArc(withCenter: NSPoint(x: x0 + w - r, y: y0 + r), radius: r, startAngle: -90, endAngle: 0)
    p.line(to: NSPoint(x: x0 + w, y: y0 + t - r)); p.appendArc(withCenter: NSPoint(x: x0 + w - r, y: y0 + t - r), radius: r, startAngle: 0, endAngle: 90)
    p.line(to: NSPoint(x: x0 + t + r * 0.0, y: y0 + t))
    p.line(to: NSPoint(x: x0 + t, y: y0 + h - r)); p.appendArc(withCenter: NSPoint(x: x0 + t - r, y: y0 + h - r), radius: r, startAngle: 0, endAngle: 90)
    p.line(to: NSPoint(x: x0 + r, y: y0 + h)); p.appendArc(withCenter: NSPoint(x: x0 + r, y: y0 + h - r), radius: r, startAngle: 90, endAngle: 180)
    p.line(to: NSPoint(x: x0, y: y0 + r)); p.appendArc(withCenter: NSPoint(x: x0 + r, y: y0 + r), radius: r, startAngle: 180, endAngle: 270)
    p.close()
    if shadow { withShadow(0.25, blur: 22 * s, dy: -10 * s) { NSColor.white.setFill(); p.fill() } } else { NSColor.white.setFill(); p.fill() }
    return (y0 + t, x0 + t)
}
func bigArrow(cx: CGFloat, top: CGFloat, bottom: CGFloat, s: CGFloat, w: CGFloat = 70, color: NSColor = .white) {
    let p = NSBezierPath(); p.lineWidth = w * s; p.lineCapStyle = .round; p.lineJoinStyle = .round
    p.move(to: NSPoint(x: cx, y: top)); p.line(to: NSPoint(x: cx, y: bottom))
    let hd = 110 * s
    p.move(to: NSPoint(x: cx - hd, y: bottom + hd)); p.line(to: NSPoint(x: cx, y: bottom)); p.line(to: NSPoint(x: cx + hd, y: bottom + hd))
    withShadow(0.2, blur: 16 * s, dy: -8 * s) { color.setStroke(); p.stroke() }
}

// V1. 메뉴바와 동일 — ㄴ + 굵은 화살표
func drawV1(_ size: CGFloat) {
    let (rect, _, s) = bgOrange(size)
    let (top, wallX) = wallShelf(rect, s)
    let cx = (wallX + rect.maxX - 130 * s) / 2
    bigArrow(cx: cx, top: rect.maxY - 180 * s, bottom: top + 110 * s, s: s)
    NSGraphicsContext.restoreGraphicsState()
}
// 소형: V1 과 동일하되 그림자 없음, 더 굵게
func drawVSmall(_ size: CGFloat) {
    let (rect, _, s) = bgOrange(size, shadow: false)
    let (top, wallX) = wallShelf(rect, s, thick: 110, shadow: false)
    let cx = (wallX + rect.maxX - 130 * s) / 2
    let p = NSBezierPath(); p.lineWidth = 100 * s; p.lineCapStyle = .round; p.lineJoinStyle = .round
    let bottom = top + 120 * s, hd = 140 * s
    p.move(to: NSPoint(x: cx, y: rect.maxY - 190 * s)); p.line(to: NSPoint(x: cx, y: bottom))
    p.move(to: NSPoint(x: cx - hd, y: bottom + hd)); p.line(to: NSPoint(x: cx, y: bottom)); p.line(to: NSPoint(x: cx + hd, y: bottom + hd))
    NSColor.white.setStroke(); p.stroke()
    NSGraphicsContext.restoreGraphicsState()
}

let out = URL(fileURLWithPath: CommandLine.arguments[1])
try? FileManager.default.createDirectory(at: out, withIntermediateDirectories: true)
func render(_ size: CGFloat, _ body: (CGFloat) -> Void) -> NSBitmapImageRep {
    let rep = NSBitmapImageRep(bitmapDataPlanes: nil, pixelsWide: Int(size), pixelsHigh: Int(size), bitsPerSample: 8, samplesPerPixel: 4, hasAlpha: true, isPlanar: false, colorSpaceName: .deviceRGB, bytesPerRow: 0, bitsPerPixel: 0)!
    NSGraphicsContext.saveGraphicsState()
    NSGraphicsContext.current = NSGraphicsContext(bitmapImageRep: rep)
    NSGraphicsContext.current?.imageInterpolation = .high
    body(size)
    NSGraphicsContext.restoreGraphicsState()
    return rep
}
let files: [(String, CGFloat)] = [
    ("icon_16x16@1x", 16), ("icon_16x16@2x", 32), ("icon_32x32@1x", 32), ("icon_32x32@2x", 64),
    ("icon_128x128@1x", 128), ("icon_128x128@2x", 256), ("icon_256x256@1x", 256), ("icon_256x256@2x", 512),
    ("icon_512x512@1x", 512), ("icon_512x512@2x", 1024),
]
for (name, px) in files {
    let rep = render(px, px <= 64 ? drawVSmall : drawV1)
    try! rep.representation(using: .png, properties: [:])!.write(to: out.appendingPathComponent(name + ".png"))
}
print("done")
